cmake_minimum_required(VERSION 3.16)
project(nd2_native LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
add_executable(nd2-native main.cpp)
option(ND2_BLAS "Use system CBLAS GEMM" OFF)
if(ND2_BLAS)
  find_package(BLAS REQUIRED)
  target_compile_definitions(nd2-native PRIVATE ND2_USE_BLAS=1)
  target_link_libraries(nd2-native PRIVATE BLAS::BLAS)
endif()
if(MSVC)
  target_compile_options(nd2-native PRIVATE /O2)
else()
  target_compile_options(nd2-native PRIVATE -O3 -ffp-contract=off)
endif()
